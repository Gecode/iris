cmake_minimum_required(VERSION 3.21)
project(iris_gecode_adapter LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Gecode CONFIG REQUIRED COMPONENTS instance)
add_executable(iris-gecode-driver main.cpp json.cpp errors.cpp)
target_compile_features(iris-gecode-driver PRIVATE cxx_std_17)
target_link_libraries(iris-gecode-driver PRIVATE Gecode::gecodeinstance)
target_compile_options(iris-gecode-driver PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(iris-gecode-checks checks.cpp json.cpp errors.cpp)
  target_compile_features(iris-gecode-checks PRIVATE cxx_std_17)
  target_link_libraries(iris-gecode-checks PRIVATE Gecode::gecodeinstance)
  target_compile_options(iris-gecode-checks PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME adapter-checks COMMAND iris-gecode-checks
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../spec/0.1/errors.md")
endif()
